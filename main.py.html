# requirements: fastapi uvicorn openai finnhub-python apscheduler python-dotenv
import os, finnhub, time
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from apscheduler.schedulers.background import BackgroundScheduler
from openai import OpenAI
from datetime import datetime
import json

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

# --- CONFIG: PUT YOUR KEYS IN .env ---
FINNHUB_KEY = os.getenv("FINNHUB_KEY", "YOUR_FINNHUB_KEY")
OPENAI_KEY = os.getenv("OPENAI_KEY", "YOUR_OPENAI_KEY")

finnhub_client = finnhub.Client(api_key=FINNHUB_KEY)
openai_client = OpenAI(api_key=OPENAI_KEY)

SIGNALS_DB = []  # Replace with Supabase / Firebase later
TICKERS = ["AAPL", "NVDA", "TSLA", "MSFT", "SPY", "AMZN", "META", "GOOGL"]

def analyze_stock(ticker):
    try:
        quote = finnhub_client.quote(ticker)
        news = finnhub_client.company_news(ticker, _from=datetime.now().strftime("%Y-%m-%d"), to=datetime.now().strftime("%Y-%m-%d"))[:3]
        news_text = " ".join([n['headline'] for n in news])[:1000]
        
        # Simple technicals (you can add RSI with yfinance)
        price = quote['c']
        
        prompt = f"""
        You are an elite stock analyst AI. Analyze {ticker}:
        Current Price: {price}
        Day Change: {quote['d']} ({quote['dp']}%)
        Recent News: {news_text}
        
        Combine: technicals (assume price action + volume), news sentiment, momentum.
        Output JSON only: {{"ticker": "{ticker}", "action": "BUY/SELL/HOLD", "confidence": 0-100, "entry": price, "tp1": price*1.02, "tp2": price*1.04, "sl": price*0.98, "timeframe": "Scalp/Day/Swing", "reason": "2 lines combining news + technicals", "sources": ["Finnhub", "News", "Technicals"]}}
        """
        
        resp = openai_client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        data = json.loads(resp.choices[0].message.content)
        data['timestamp'] = datetime.now().isoformat()
        data['company'] = ticker
        return data
    except Exception as e:
        return {"ticker": ticker, "action": "HOLD", "confidence": 50, "reason": f"Error: {e}", "timestamp": datetime.now().isoformat()}

def job_every_10_min():
    print(f"[{datetime.now()}] AI Scan Starting...")
    new_signals = []
    for t in TICKERS:
        sig = analyze_stock(t)
        new_signals.append(sig)
        time.sleep(1)  # rate limit
    global SIGNALS_DB
    SIGNALS_DB = new_signals + SIGNALS_DB[:50]  # keep last 50
    print(f"Generated {len(new_signals)} signals")

scheduler = BackgroundScheduler()
scheduler.add_job(job_every_10_min, 'interval', minutes=10)
scheduler.start()
job_every_10_min()  # run on startup

@app.get("/api/signals")
def get_signals():
    return {"signals": SIGNALS_DB, "next_scan_in": "10:00", "engine": "ACTIVE", "sources": ["Finnhub", "SEC", "FinBERT", "TradingView", "News", "Volume AI"]}


@app.get("/widget")
def widget():
    from fastapi.responses import HTMLResponse
    # Simple widget HTML that fetches /api/signals
    html = """
    <html><head><style>
    body{background:#0a0a0f;color:white;font-family:monospace;padding:20px}
    .card{background:#15151f;border:1px solid #222;border-radius:12px;padding:16px;margin:12px 0}
    .buy{color:#00ff88}.sell{color:#ff4444}
    </style></head><body>
    <h2>⚡ Huxla AI Engine - LIVE (10 min scan)</h2>
    <div id="feed">Loading AI signals...</div>
    <script>
    async function load(){
      const r = await fetch('/api/signals');
      const data = await r.json();
      document.getElementById('feed').innerHTML = data.signals.map(s=>`
        <div class='card'>
          <b>${s.ticker}</b> <span class='${s.action.includes('BUY')?'buy':'sell'}'>${s.action}</span> - ${s.confidence}%<br>
          <small>${s.reason||''}</small><br>
          Entry $${s.entry} | TP1 $${s.tp1} | SL $${s.sl} | ${s.timestamp}
        </div>
      `).join('');
    }
    load(); setInterval(load, 600000);
    </script></body></html>
    """
    return HTMLResponse(content=html)

@app.get("/")
def root():
    return {"status": "Huxla AI Engine Running", "endpoints": ["/api/signals"]}
